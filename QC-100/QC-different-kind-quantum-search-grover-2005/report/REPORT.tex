\begin{abstract}
We independently replicate the central results of Grover's 2005 paper
\emph{A different kind of quantum search} in a pure-\texttt{numpy} statevector
simulator. The paper's key identity
$\|U_{m,ts}\|^2 = 1 - \epsilon^{3^m}$ is reproduced to machine precision
($\sim 10^{-14}$ absolute error) for $m = 0, \dots, 4$; the $\pi/3$ recursion
is empirically monotone; and standard Grover shows the expected oscillation
for the same problem instance ($N = 16$, one marked element). LLM-judge
verdict (Argo free endpoint, \texttt{argo:gpt-4o}, $T = 0$): REPLICATED,
coverage 1.0, agreement 1.0.
\end{abstract}

\section{Paper Summary}
Grover shows that replacing the two selective inversions (phase $\pi$) at the
heart of amplitude amplification with selective \emph{phase shifts} of $\pi/3$
turns the algorithm into a \emph{fixed-point} search: instead of oscillating
around $|t\rangle$ like standard Grover, the state vector converges monotonically
towards $|t\rangle$. Formally, if $U$ maps $|s\rangle$ to $|t\rangle$ with base
probability $\|U_{ts}\|^2 = 1 - \epsilon$, then:
\begin{itemize}
  \item Single-shot: $\langle t | U R_s U^\dagger R_t U | s \rangle$ has squared
        magnitude $1 - \epsilon^3$ (Section 3, Eq.\ derived).
  \item Recursion (Section 4): with $U_0 = U$ and
        $U_{m+1} = U_m R_s U_m^\dagger R_t U_m$, one has
        $\|U_{m,ts}\|^2 = 1 - \epsilon^{3^m}$, using $q_m = 3^m$ base $U$-calls.
\end{itemize}
Two applications are discussed: (i) quantum search when the number of marked
items is uncertain (Section 5), and (ii) systematic-error correction without
syndrome extraction (Section 6).

\section{Method}
\begin{enumerate}
  \item \textbf{Environment.} macOS on CherryRd; Python 3.13 in a fresh venv;
        \texttt{numpy 2.5.1}, \texttt{matplotlib 3.11.0}. Free Argo endpoint
        (\url{http://127.0.0.1:44497/v1}), key \texttt{stevens}, model
        \texttt{argo:gpt-4o} used only for LLM-judge scoring.
  \item \textbf{PDF fetch.} \texttt{curl -sL -o paper.pdf https://arxiv.org/pdf/quant-ph/0503205} (138 kB, 13 pp.).
  \item \textbf{Text extraction.} Marker (\texttt{extraction/marker.md}) and
        Nougat (\texttt{extraction/nougat.mmd}) pulled from the central QC-200
        parsed corpus (already produced 2026-07-05).
  \item \textbf{Statevector construction.} $n = 4$ qubits, $N = 16$. Built
        $W = H^{\otimes n}$ via numpy Kronecker; phase-shift operators
        $R_x(\varphi)$ diagonal with $e^{i\varphi}$ on index $x$; inversion
        $I_x = R_x(\pi)$. Full complex-\texttt{numpy} 16-dim state.
  \item \textbf{Standard Grover.} $G = W I_0 W I_t$; iterate on
        $|s\rangle = W|0\rangle$; record $|\langle t | \text{state}\rangle|^2$
        for $k = 0, \dots, 12$.
  \item \textbf{$\pi/3$ recursion.} $U_0 = W$;
        $U_{m+1} = U_m R_s U_m^\dagger R_t U_m$;
        $R_s = R_0(\pi/3)$, $R_t = R_{\text{target}}(\pi/3)$; apply to
        $|0\rangle$; record $|\langle t|U_m|0\rangle|^2$ for $m = 0, \dots, 4$;
        compare to $1 - \epsilon^{3^m}$ where $\epsilon = 1 - 1/N = 15/16$.
  \item \textbf{Monotonicity.} \texttt{np.diff(P) $\geq -10^{-12}$}.
  \item \textbf{LLM-judge scoring.} Argo \texttt{argo:gpt-4o}, $T = 0$,
        prompt = full claim list + full numeric evidence; JSON verdict schema
        (\texttt{claim\_scores, coverage\_fraction, agreement\_fraction,
        verdict, one\_line, justification}). No regex on paper text.
  \item \textbf{Artifacts.} All 8 mandatory files under the target directory;
        evidence outputs (JSON, PNG, logs) under \texttt{report/evidence/}.
\end{enumerate}

\section{Results vs Paper}

\subsection{Base probability}
Theory: $\|U_{ts}\|^2 = 1/N = 0.0625$. Measured: $0.062500$. \emph{Exact.}

\subsection{Standard Grover --- oscillation (C4)}
For $k = 0, \dots, 12$:
\[
P(k) \in \{0.062, 0.473, 0.908, 0.961, 0.582, 0.126, 0.020,
0.365, 0.836, 0.992, 0.687, 0.206, 0.001\}.
\]
Peak at $k = 9$ ($P \approx 0.992$); dips to $\approx 0.001$ by $k = 12$
(``burnt souffl\'e''). The $\pi/3$ recursion by contrast climbs monotonically
$0.063 \to 0.176 \to 0.441 \to 0.825 \to 0.995$.

\subsection{Reproduced figures}
\texttt{report/evidence/fig\_probability\_trajectory.png} plots both curves
against oracle-query count (paper Figure 1 analogue: oscillation vs monotone
staircase). \texttt{report/evidence/fig\_failure\_scaling.png} plots
$\log(1 - P)$ vs $m$ overlaid with the analytic $\epsilon^{3^m}$; perfect
overlay.

\section{Per-Claim ``What Worked / What Didn't''}
\begin{description}
  \item[C1 (monotone convergence).] \textbf{Worked.} Diff of successive
        $P_m$ values is non-negative to $10^{-16}$. No tuning required.
  \item[C2 (single-shot cubic identity).] \textbf{Worked exactly.} $m=1$
        row matches theory to $4\times 10^{-16}$.
  \item[C3 (triple-exponent recursion).] \textbf{Worked exactly through $m=4$.}
        At $m=5$ the base $U$-call count would be $243$; the operator remains
        $16\times 16$ so it would still be trivial, but $\epsilon^{3^5}$ is
        already below double-precision resolution (would round to 0), which
        is why we stopped at $m=4$.
  \item[C4 (std-Grover oscillation).] \textbf{Worked qualitatively and
        quantitatively.} The observed peaks at $k=3, 9$ match the standard
        analytic period $\pi / (2\theta)$ with $\theta = \arcsin(\sqrt{1/N})$.
  \item[C5 (systematic-error correction).] \textbf{Not tested.} Would require
        a noise model (Kraus operators / decoherence channel); paper does not
        report a numerical simulation either. Out of scope for the requested
        figure.
  \item[C6 (search-amidst-uncertainty benchmark).] \textbf{Not tested.} The
        replication brief asked specifically for the probability-trajectory
        figure; the uncertainty benchmark is a separate result that could be
        added in a follow-up.
\end{description}
