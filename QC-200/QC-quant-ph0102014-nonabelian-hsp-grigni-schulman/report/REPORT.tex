\section{Paper Summary}

The paper shows that HSP is quantum-polynomial-time for four important
classes of non-Abelian groups:
\begin{enumerate}
  \item All \emph{normal} HSP in solvable and permutation groups (Thm.~8),
        by combining Beals--Babai classical black-box-group machinery with
        quantum order-finding.
  \item HSP in groups whose commutator subgroup $G'$ is small
        (Thm.~11), giving \textsc{polytime}$+\lvert G'\rvert$.
  \item Extra-special $p$-groups (Cor.~12), a special case of (2).
  \item Groups with an elementary Abelian normal 2-subgroup $N$ (given by
        generators) whose quotient $G/N$ is small or cyclic (Thm.~13).
        This is the paper's main new construction, containing the wreath
        products $\mathbb{Z}_2^k \wr \mathbb{Z}_2$ (Rötteler--Beth) and
        dihedral-like groups as special cases.
\end{enumerate}

The pivotal quantum primitive is \textbf{Lemma 9}: Abelian HSP with a
\emph{quantum-state} oracle (each $\lvert f(g)\rangle$ a unit vector,
constant on left cosets of $H$, orthogonal across distinct cosets) is
polytime by the standard 4-step algorithm
\[
\lvert 1_A\rangle\lvert 0^m\rangle
\xrightarrow{\text{QFT}_A} \frac{1}{\sqrt{|A|}}\sum_g \lvert g\rangle\lvert 0^m\rangle
\xrightarrow{f} \frac{1}{\sqrt{|A|}}\sum_g \lvert g\rangle\lvert f(g)\rangle
\xrightarrow{\text{QFT}_A} \sum_{g \in A/H,\ h \in H^\perp} \chi_h(g)\lvert h\rangle\lvert f(g)\rangle
\rightarrow \text{measure}.
\]
The output register measurement gives a uniform sample from $H^\perp$.

\subsection{Lemma 9 verification (claim C1)}

\textbf{Driver:} \texttt{work/lemma9\_verify.py}.

For $n$-bit groups $A = (\mathbb{Z}_2)^n$ with $n \in \{2,3,4,5\}$ and
seeds $\{1,\ldots,5\}$:
\begin{enumerate}
  \item Sample a random $F_2$-subgroup $H \le A$.
  \item Construct orthonormal coset states $\lvert\psi_C\rangle$ in
        $d = \lceil\log_2(|A|/|H|)\rceil$ qubits by unitary-conjugating the
        identity's leading columns with a random $Q$ from a complex
        QR decomposition. This is a \emph{genuine quantum-state} oracle,
        not just a classical coset-label.
  \item Build the block-diagonal unitary $U_f \lvert x\rangle\lvert 0\rangle
        = \lvert x\rangle\lvert\psi_{\mathrm{coset}(x)}\rangle$ and
        Gram--Schmidt-complete the remaining columns to unitarity
        (checked to $\lVert U^\dagger U - I \rVert \le 10^{-8}$).
  \item Run the exact circuit
        $H^{\otimes n} \cdot U_f \cdot H^{\otimes n}$ on
        $(n + d)$ qubits via
        \texttt{qiskit.quantum\_info.Statevector.from\_instruction};
        marginalise the input register.
  \item Compare $\mathrm{Prob}(w)$ against the analytic prediction
        $1/|H^\perp|$ for $w \in H^\perp$, zero otherwise.
\end{enumerate}

\textbf{Result:} 20/20 trials pass with
$\max_{w \in H^\perp}|p_w - 1/|H^\perp|| \le 1.3\times 10^{-15}$ and
$\max_{w \notin H^\perp} p_w = 0$ exactly.

\subsection{Theorem 13 replication (claim C5)}

\textbf{Driver:} \texttt{work/hsp\_ims\_theorem13.py}.

\textbf{Group model.} $G = \mathbb{Z}_2^{2k} \rtimes \mathbb{Z}_2$
(the base of the wreath product $\mathbb{Z}_2^k \wr \mathbb{Z}_2$),
with $\sigma \in \mathrm{Aut}(N)$ swapping the top-$k$ and bottom-$k$
bits. Multiplication
\[
  (x_1, s_1) \cdot (x_2, s_2) = (x_1 \oplus \sigma^{s_1}(x_2),\ s_1 \oplus s_2).
\]
$N = \mathbb{Z}_2^{2k}$ is a genuine elementary Abelian normal 2-subgroup;
$|G/N| = 2$ (cyclic $\Rightarrow$ Theorem 13's polytime case).

\textbf{Planted subgroups.} For each trial,
\texttt{random\_hidden\_subgroup} picks 1--2 random group elements as
generators and takes the closure. Trials alternate:
\begin{itemize}
  \item \textbf{intersect\_z\_coset=True} --- the first generator has second
        coordinate $s = 1$ (exercises Step B).
  \item \textbf{intersect\_z\_coset=False} --- $H \subseteq N$ (only Step A).
\end{itemize}

\textbf{Oracle.} \texttt{build\_coset\_oracle} labels every element of $G$
by its left-coset index using the planted $H$; oracle queries reduce to
$O(1)$ table lookups.

\textbf{Algorithm (Ivanyos--Magniez--Santha Theorem 13, verbatim).}
\begin{itemize}
  \item \textbf{Step A.} Restrict $f$ to $N$. Run Abelian HSP on
        $N \simeq (\mathbb{Z}_2)^{2k}$ to recover $H \cap N$.
  \item \textbf{Step B.} For the single nontrivial coset representative
        $z = (0,1)$ of $G/N$, define
        $F : \mathbb{Z}_2 \times N \to Y$ by $F(0,x) = f((x,0))$ and
        $F(1,x) = f((x,0)\cdot z)$. Run Abelian HSP on
        $\mathbb{Z}_2 \times N \simeq (\mathbb{Z}_2)^{2k+1}$.
        Any generator with first bit $=1$ has the form $(1, u)$ with
        $u \cdot z \in H$ (using 2-torsion of $N$ to identify $u = u^{-1}$).
  \item \textbf{Step C.} $H_1 := \langle H\cap N,\ \{u\cdot z\}\rangle$;
        enumerate under group multiplication and verify $H_1 = H_{\text{planted}}$
        (set equality).
\end{itemize}

\textbf{Two computation paths.} For $n_\text{input} + n_\text{output} \le 8$
qubits (only $k=1$), the sampler builds the full unitary oracle and runs
\texttt{AerSimulator(method='statevector')} with \texttt{shots=1}. For
larger cases, the sampler evaluates the exact analytic distribution
\[
  \mathrm{Prob}(w) = \frac{1}{|A|^2}
    \sum_{\text{coset } C \subseteq A}
    \Big| \sum_{x \in C} (-1)^{w \cdot x} \Big|^2
\]
and draws from it. The two paths were cross-checked on the $k=1$ case; both
produce statistically indistinguishable behaviour.

\textbf{Sweep.} $k \in \{1,2,3,4\}$ ($|G| \in \{8, 32, 128, 512\}$),
6 trials each, seed 20260706.
