\section{Paper's claims (as replicated)}
\begin{enumerate}
\item[(P1)] The undistorted G-type AFM KNiF$_3$ has \emph{no} spin splitting (type-IV AFM;
Kramers degeneracy protected by $\mathcal{PT}$ and $\tau\mathcal{U}_{1/2}$).
\item[(P2)] The $a^0b^-c^-$ distortion (rotation about $y$) produces a \emph{d-wave}
altermagnetic spin-splitting pattern in the $k_z{=}0$ plane.
\item[(P3)] The $a^0b^0c^-$ distortion (out-of-phase rotation about $z$, $\eta_z\approx1$)
produces a \emph{g-wave} pattern.
\item[(P4)] The splitting is \emph{switched on} by the symmetry-breaking rotation:
$\eta=0$ (in-phase) gives no splitting, $\eta=1$ (out-of-phase) gives maximal splitting.
\item[(P5)] Time evolution shows \emph{mode switching} (Fig.~2b): dominant g-wave
($z$-rotation, early) $\leftrightarrow$ d-wave ($y$-rotation, late).
\item[(P6)] An anomalous Hall conductivity emerges (Fig.~3): zero in the ground state,
finite ($\sigma_{xy}$ peaks $\sim\pm\SI{400}{S/cm}$) once distorted; its evaluation needs SOC.
\end{enumerate}

\section{Governing equations reimplemented}
The minimal altermagnet surrogate on the (sublattice $A/B$)$\,\otimes\,$(spin) basis, at
$k_z=0$:
\begin{equation}
H(\mathbf{k}) = \varepsilon_0(\mathbf{k})\,\tau_0\sigma_0
+ f(\mathbf{k})\,\tau_x\sigma_0
+ J\,\tau_z\sigma_z
+ \eta\,g_{\text{mode}}(\mathbf{k})\,\tau_z\sigma_0 ,
\end{equation}
with $f(\mathbf{k})=-2t\cos\tfrac{k_x}{2}\cos\tfrac{k_y}{2}$ (inter-sublattice NN hopping),
$\varepsilon_0=-2t'(\cos k_x+\cos k_y)$, and the N\'eel exchange $J\,\tau_z\sigma_z$ (no SOC).
The altermagnet term is a \emph{sublattice-staggered, spin-independent} anisotropic
hopping $g_{\text{mode}}(\mathbf{k})\,\tau_z\sigma_0$ whose interplay with $J\,\tau_z\sigma_z$
splits the spin-up and spin-down blocks ($E_\uparrow$ sees $(J+\eta g)\tau_z$,
$E_\downarrow$ sees $(-J+\eta g)\tau_z$)---the standard collinear-altermagnet mechanism:
\begin{align}
g_{d\text{-wave}}(\mathbf{k}) &= t_{am}(\cos k_x - \cos k_y), &(m=2)\\
g_{g\text{-wave}}(\mathbf{k}) &= t_{am}\,\sin k_x \sin k_y (\cos k_x - \cos k_y). &(m=4)
\end{align}
The spin-splitting map is $\Delta(\mathbf{k})=E_\uparrow(\mathbf{k})-E_\downarrow(\mathbf{k})$
(VBM, $k_z{=}0$). $\eta\in[0,1]$ is the paper's symmetry-breaking parameter,
$\eta_\alpha=1-\tfrac{|\theta_\alpha^u-\theta_\alpha^l|}{2\max(|\theta_\alpha^u|,|\theta_\alpha^l|)}$.
For the Hall check (P6) a small distortion-locked SOC $\propto\eta\sin\tfrac{k_x}{2}\sin\tfrac{k_y}{2}\,\tau_y\sigma_z$
is added; it vanishes in the ground state and turns on with the altermagnet order.

\section{Methods}
\textbf{Part A (statics --- spin splitting).} Diagonalize $H(\mathbf{k})$ on a $48\times48$
$k_z{=}0$ grid; form $\Delta(\mathbf{k})$; Fourier-decompose on a constant-$|\mathbf{k}|$
loop to extract the dominant angular harmonic $m$ and count sign nodes; test parity and
the $k_x\!\leftrightarrow\!k_y$ mirror relation.
\textbf{Part B (transport --- Hall).} Kubo--Bastin occupied-state sum over states with
$v_a=\partial H/\partial k_a$ and generalized current $j^{S_z}_x=\tfrac12\{S_z,v_x\}$
(spin Hall) or $v_x$ (charge). The Kubo--Bastin Hall machinery is adapted from and credited to
the shared kernel \texttt{gobel2024\_sd\_skyrmion\_kubo\_Lz\_kernel.py}
(G\"obel \textit{et al.} 2024, arXiv:2410.00820).

\texttt{crux} compute target. Here $\eta$ is an external control knob, so mode switching
(C5) is reproduced as a \emph{symmetry statement}, not a fs time trace.
\item \textbf{Net charge AHC ($\pm$400 S/cm) is symmetry-compensated to zero in the minimal
2D collinear surrogate.} The nonrelativistic Hall signature we \emph{do} reproduce is the
\emph{spin} Hall (zero in ground state, finite once distorted). The absolute $\pm400$ S/cm

\item \textbf{Machine-precision residuals are expected.} The surrogate obeys the parity /
mirror relations by construction ($\sin k$ exactly odd; $\cos k_x-\cos k_y$ exactly
mirror-odd), so residuals $\sim10^{-16}$ are not suspicious---the evidence is the
simultaneous multi-claim match plus the genuine failure of a wrong parameterization.
\end{enumerate}
