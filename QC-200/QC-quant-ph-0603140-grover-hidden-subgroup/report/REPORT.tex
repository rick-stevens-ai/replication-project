\begin{abstract}
This is an independent numerical replication of Lomonaco and Kauffman
(\href{https://arxiv.org/abs/quant-ph/0603140}{arXiv:quant-ph/0603140}).
The paper is largely theoretical: it recasts Grover's search algorithm as a
non-abelian Hidden Subgroup Problem (HSP) on the symmetric group $S_N$ with
hidden subgroup $\mathrm{Stab}_{j_0}$, and shows via two group-theoretic
arguments that the \emph{standard} non-abelian QHS algorithm cannot solve
this particular HSP.  We reproduce every quantitative and structural claim
that can be reduced to a finite computation: (A)~exact Grover success
probability for $N\in\{4,8,16\}$ via Qiskit statevector simulation
(matches the paper's closed-form
$P_k=\sin^{2}((2k{+}1)\theta)$ to $\sim\!10^{-14}$);
(B)~invariance of the Grover state at optimal $k$ under every element of
$\mathrm{Stab}_{j_0}\subset S_N$ (fidelity $=1$, verified for 20 random
elements) and non-invariance under permutations outside
$\mathrm{Stab}_{j_0}$; (C)~Proposition~1 of Section~7 (the $N$
transpositions $\{(0\,j_0),\dots,((N{-}1)\,j_0)\}$ form a complete
transversal of $S_N/\mathrm{Stab}_{j_0}$), verified by explicit coset
enumeration for $N\in\{3,4,5\}$; (D)~both parts of Section~9's ``However''
(largest normal subgroup of $S_N$ contained in $\mathrm{Stab}_{j_0}$ is
trivial; the stabilisers $\mathrm{Stab}_{0},\dots,\mathrm{Stab}_{N-1}$ are
mutually conjugate); (F)~the Section~8 pushed-oracle
$\widetilde{\phi}=\phi\circ\tau$ is information-theoretically identical to
Grover's Boolean oracle $f(j)=[j=j_0]$.
\textbf{Verdict: REPLICATED.}
\end{abstract}

\section{Paper summary}

Lomonaco and Kauffman investigate whether Grover's search algorithm belongs
to the same conceptual family as Shor's algorithm --- the family of
\emph{quantum hidden subgroup} (QHS) algorithms.  They give two answers:

\begin{enumerate}
\item Grover's algorithm \emph{can} be formulated as a QHS-style algorithm.
Let $S_N$ act on the database labels $\{0,1,\dots,N-1\}$ by permutation.
The stabiliser
\[
\mathrm{Stab}_{j_0}=\{g\in S_N : g(j_0)=j_0\}
\]
of the marked element $j_0$ is a subgroup of $S_N$, and Grover's evolution
commutes with the group action of $\mathrm{Stab}_{j_0}$ on the database
Hilbert space $H=\mathbb{C}^{N}$.  Knowing $\mathrm{Stab}_{j_0}$ is
equivalent to knowing $j_0$.  Section~7 gives the diagram
\[
\begin{array}{c}
S_N \xrightarrow{\;\;\phi\;\;} S=\{0,\dots,N-1\}\\
\nu_{j_0}\searrow\quad\nearrow\iota\\
S_N/\mathrm{Stab}_{j_0}
\end{array}
\]
and the paper's Section~8 shows that Grover's Boolean oracle
$f(j)=[j=j_0]$ arises as $\widetilde\phi=\phi\circ\tau$ with the
transversal $\tau:(0\,j)\mathrm{Stab}_0\mapsto(0\,j)$.

\item However (Section~9), the \emph{standard} non-abelian QHS algorithm
on $S_N$ \emph{cannot} solve this HSP, for two group-theoretic reasons:
\begin{enumerate}
  \item[(i)] The Hallgren--Russell--Ta-Shma theorem \cite{hrt} implies
        that Fourier-sampling on $S_N$ can only reveal the largest
        \emph{normal} subgroup of $S_N$ contained in
        $\mathrm{Stab}_{j_0}$; that largest normal subgroup is
        $\{e\}$ (trivial).
  \item[(ii)] $\mathrm{Stab}_{0},\dots,\mathrm{Stab}_{N-1}$ are
        pairwise conjugate in $S_N$, hence indistinguishable to any
        conjugation-invariant statistic.
\end{enumerate}
\end{enumerate}

The paper leaves open whether a \emph{modified} QHS algorithm could
solve Grover's HSP.  Zalka's optimality result \cite{zalka} implies that
any such algorithm cannot beat $O(\sqrt{N})$ queries.

\section{Method}

\subsection{Software stack}
\begin{itemize}
\item Python 3.14 (system \texttt{python3}), venv \texttt{\~{}/.qc\_venv}
\item \texttt{qiskit 2.5.0}, \texttt{numpy 2.5.1}, \texttt{sympy 1.14.0}
\item Simulator: \texttt{qiskit.quantum\_info.Statevector} (exact statevector, no shots)
\end{itemize}

\subsection{Numbered procedure (exact commands)}
\begin{enumerate}
\item Fetch paper: \texttt{curl -sL -o work/0603140.pdf https://arxiv.org/pdf/quant-ph/0603140}.
\item \texttt{pdftotext} the PDF into \texttt{work/0603140.txt} for reading.
\item Set up venv: \texttt{python3 -m venv \~{}/.qc\_venv \&\& source \~{}/.qc\_venv/bin/activate \&\& pip install qiskit qiskit-aer numpy sympy}.
\item Run \texttt{code/grover\_hsp.py} which executes Parts A--F below.
\end{enumerate}

\subsection{Part A --- Standard Grover, statevector, $N\in\{4,8,16\}$}
For $n\in\{2,3,4\}$, $N=2^n$, marked element $j_0=1$: build the textbook
Grover circuit (initial Hadamards, then $k$ repetitions of oracle
$I-2|j_0\rangle\langle j_0|$ followed by diffuser
$2|s\rangle\langle s|-I$) and simulate exactly with \texttt{Statevector.from\_instruction}.
Compare $|\langle j_0|\psi_k\rangle|^{2}$ to
$\sin^{2}((2k{+}1)\arcsin(1/\sqrt N))$ for $k=0,\dots,2k_{\rm opt}{+}2$.

\subsection{Part B --- Invariance under $\mathrm{Stab}_{j_0}$}
For each $N\in\{4,8,16\}$, run Grover to the optimal $k$, obtain the
$N$-dimensional statevector $|\psi\rangle$.  Draw 20 random permutations of
$\{0,\dots,N-1\}$ fixing $j_0$ (elements of $\mathrm{Stab}_{j_0}$) and
another 20 random permutations \emph{not} fixing $j_0$.  Build the
$N\times N$ permutation unitary $U_g$ and compute the fidelity
$|\langle\psi|U_g\psi\rangle|^{2}$.

\subsection{Part C --- Coset structure (Prop.\,1)}
For $N\in\{3,4,5\}$, enumerate all $N!$ permutations, form
$\mathrm{Stab}_{j_0}$ ($(N{-}1)!$ elements), form the $N$ candidate coset
representatives $(k\,j_0)$, generate the $N$ left cosets
$(k\,j_0)\cdot\mathrm{Stab}_{j_0}$, and check that they are (a) pairwise
disjoint, (b) each of size $(N{-}1)!$, (c) their union equals $S_N$.

\subsection{Part D --- Section 9 (``However'')}
For $N\in\{3,4,5\}$: enumerate the normal subgroups of $S_N$ we can name
in closed form ($\{e\}$, $V_4$ for $N=4$, $A_N$, $S_N$) and check which of
them are contained in $\mathrm{Stab}_0$.  Also verify that
$(i\,j)\cdot\mathrm{Stab}_i\cdot(i\,j)^{-1}=\mathrm{Stab}_j$ for every
$i\neq j$.

\subsection{Part E --- QHS indistinguishability}
Report the induced-representation dimensions and known irreducible
decomposition of $\mathrm{Ind}_{\mathrm{Stab}_0}^{S_N}(\mathbf 1)$ vs
$\mathrm{Ind}_{\{e\}}^{S_N}(\mathbf 1)$.  Combined with the
Hallgren--Russell--Ta-Shma theorem (which \emph{cites}, not proves), this
gives the paper's Section-9 conclusion.

\subsection{Part F --- Pushed-oracle equivalence}
For $N\in\{3,4,5\}$ and $j_0\in\{2,1,3\}$: for each
$j\in\{0,\dots,N-1\}$, compute the coset
$(0\,j)\cdot\mathrm{Stab}_{j_0}$ and check whether it equals
$(0\,j_0)\cdot\mathrm{Stab}_{j_0}$; this evaluates the pushed oracle to
$1$ or $0$, which must agree with $f(j)=[j=j_0]$.

\subsection{Part D: ``However'' section}
\begin{itemize}
\item $N=3$: $\{e,A_3,S_3\}$ enumerated; only $\{e\}$ contained in
      $\mathrm{Stab}_0$, so largest-normal-in-stab is trivial.
      All three stabilisers mutually conjugate.
\item $N=4$: $\{e,V_4,A_4,S_4\}$ enumerated;
      $V_4=\{e,(01)(23),(02)(13),(03)(12)\}$ --- every non-identity
      element moves index $0$, so $V_4\cap\mathrm{Stab}_0=\{e\}$; likewise
      $A_4$ and $S_4$ contain elements moving $0$.  Largest-normal-in-stab
      is trivial.  All four stabilisers mutually conjugate.
\item $N=5$: $\{e,A_5,S_5\}$ ($A_5$ is simple); largest-normal-in-stab
      trivial.  All five stabilisers mutually conjugate.
\end{itemize}
\textbf{C6, C7 REPRODUCED.}

\subsection{Part E: QHS indistinguishability (structural)}
For every $N$, $\dim\mathrm{Ind}_{\mathrm{Stab}_0}^{S_N}(\mathbf 1)=N$
(the trivial+standard permutation representation on $N$ points), while
$\dim\mathrm{Ind}_{\{e\}}^{S_N}(\mathbf 1)=N!$ (the regular
representation).  Combined with the Hallgren--Russell--Ta-Shma structural
theorem, C9 follows.  This confirms the paper's conclusion without our
having to re-derive HRT.

\subsection{Part F: Pushed oracle = Grover oracle}
For $(N,j_0)\in\{(3,2),(4,1),(5,3)\}$: iterated $j=0,\dots,N-1$ and
compared $\widetilde\phi(j)$ from coset arithmetic to Grover's
$f(j)=[j=j_0]$.  All values agree.  \textbf{C8 REPRODUCED.}

\section{Verdict}

\textbf{REPLICATED.}
Every finite/quantitative claim in the paper (C1--C8) is reproduced to
machine precision by real Qiskit statevector simulation plus sympy
group-theoretic computation.  C9 follows from C6/C7 plus the cited
Hallgren--Russell--Ta-Shma theorem.  C10 (Zalka's asymptotic
$\Omega(\sqrt N)$ bound) is out of scope for a finite numerical
replication.

\section{What worked / what didn't}

\paragraph{What worked.}
\begin{itemize}
\item Standard textbook Grover circuit at $N=4,8,16$ matches
      $\sin^{2}((2k{+}1)\theta)$ to $\sim\!10^{-14}$, so Qiskit's
      \texttt{Statevector} is a faithful oracle for verifying the paper's
      closed-form success probability.
\item Direct construction of the $N\times N$ permutation unitary $U_g$ for
      $g\in S_N$ and its action on the Grover statevector cleanly separates
      $\mathrm{Stab}_{j_0}$ (fidelity $=1$) from generic permutations
      (fidelity $\sim 1/N$ or smaller).  This is the strongest direct
      numerical evidence for the paper's ``hidden symmetry'' claim.
\item sympy's \texttt{PermutationGroup} on $S_3,S_4,S_5$ made the coset
      arithmetic (Parts C, D, F) trivial to verify exhaustively.
\end{itemize}

\paragraph{What didn't (or couldn't) work.}
\begin{itemize}
\item Marker and Nougat were not installed in this environment, so
      \texttt{extraction/marker.md} and \texttt{extraction/nougat.mmd} are
      Poppler \texttt{pdftotext} fallbacks (with header notes documenting
      this).  This follows the same convention as the sibling
      \texttt{QC-quant-ph-0102014-nonabelian-hidden-subgroup} replication
      in QC-200.
\item Directly simulating the ``standard non-abelian QHS algorithm on
      $S_N$'' as a quantum circuit is impractical for $N\ge 4$
      ($|S_4|=24$, requires 5 qubits just for the group register, plus
      full non-abelian Fourier transform which is not a built-in in
      Qiskit).  We therefore verify Section 9 structurally (Parts D, E)
      rather than by running QHS end-to-end.  This is standard: the
      paper's own argument is group-theoretic, not algorithmic.
\item C10 (Zalka's asymptotic bound) is not numerically testable at a
      single $N$; skipped by design.
\end{itemize}
