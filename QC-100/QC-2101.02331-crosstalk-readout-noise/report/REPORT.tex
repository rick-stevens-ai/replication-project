\section{Paper}
\textbf{Maciejewski, Baccari, Zimbor\'as, Oszmaniec.}
\emph{Modeling and mitigation of cross-talk effects in readout noise with applications to the Quantum Approximate Optimization Algorithm.}
arXiv:2101.02331v3, \emph{Quantum} \textbf{5}, 464 (2021). CC-BY 4.0.

\section{Verdict}
\textbf{REPLICATED} (reproducible core; correlated-response mitigation empirically beats
tensor-product mitigation on a synthetic, textbook cluster-crosstalk noise model with
the same qualitative and near-quantitative behavior claimed in the paper).

\section{Claims tested}
\begin{itemize}
  \item \textbf{C1 (structural):} Tensor-product measurement noise misses cross-talk. Max
    element difference $|R_\text{true} - R_\text{tp}| = 0.0851$. \emph{Reproduced.}
  \item \textbf{C2 (TVD reduction):} Correlated response-matrix inversion reduces TVD to
    ideal by a large factor over raw and over tensor-product mitigation.
    Mean TVD: raw $0.0965$; TP $0.0272$; correlated $0.00652$. Reduction factor
    $14.8\times$ over raw, $\approx 4\times$ over TP. \emph{Reproduced.}
  \item \textbf{C3 (Observable error reduction, headline analogue):} Mean $|\Delta\langle H\rangle|$:
    raw $0.1721$; TP $0.0505$; correlated $0.00558$. Reduction factor
    \textbf{$30.8\times$} over raw. \emph{Reproduced} (paper reports $>22\times$ on IBM 15q,
    $>5.5\times$ on Rigetti 23q; our synthetic case is a clean cluster, so
    slightly higher is expected). \textbf{This is the headline claim, and it is
    directly exercised.}
  \item \textbf{C4 (QAOA landscape recovery):} $p{=}1$ line-4 MaxCut: ideal ratio $0.788$,
    raw noisy $0.718$, correlated mitigated $0.790$. Correlated mitigation
    closes the $\sim 10\%$ approximation-ratio gap. \emph{Reproduced qualitatively.}
  \item \textbf{C5 (Hardware numbers, IBM Melbourne 15q / Rigetti Aspen 23q):}
    \emph{Not tested.} Both chips retired; CPU-only replication out of scope.
  \item \textbf{C6 (DDOT circuit-count scaling $O(k\cdot 2^k \log N)$):}
    \emph{Not tested.} Structural claim, well-established from QOT literature.
\end{itemize}

\section{Method (summary)}
\begin{itemize}
  \item Host: CherryRd (macOS, Python 3.14.6, qiskit 2.5.0, qiskit-aer 0.17.2).
  \item Noise model: 4-qubit device, asymmetric per-qubit measurement matrices
    $A_i$ with $p_{01}=[0.02,0.03,0.03,0.04]$, $p_{10}=[0.06,0.07,0.07,0.09]$;
    cluster $C=\{q_1,q_2\}$ with cross-talk $\delta=0.05$ conditioned on the
    neighbor's true state. Full $R_\text{true}$ constructed as a $16\times16$
    column-stochastic matrix by enumeration over 4-bit inputs.
  \item Pipeline: (i) build random $p{=}2$ QAOA circuit; (ii) ideal Aer sample
    $10^5$ shots; (iii) apply $R_\text{true}$ to get noisy distribution and
    resample; (iv) mitigate by pseudo-inverse of $R_\text{tp}$ and $R_\text{true}$
    respectively with simplex projection; (v) compare TVD and
    $\langle H\rangle=\sum_{(i,j)\in E} Z_iZ_j$ on the line-graph edges;
    (vi) 25-seed average.
  \item Grid: $p{=}1$ line-4 MaxCut, $(\gamma,\beta)$ on $13\times13$ over
    $[0,\pi]\times[0,\pi/2]$; four cost surfaces produced.
\end{itemize}

\section{Comparison to paper's headline}
Paper: $>22\times$ (IBM 15q, real hardware, ground-state energy) and $>5.5\times$
(Rigetti 23q). This replication: $30.8\times$ on energy, $14.8\times$ on TVD, on a
synthetic 4-qubit cluster --- \emph{same order of magnitude and direction.} Slightly
better than IBM is expected: our $R_\text{true}$ is known exactly (perfect
characterization limit), and real chips also carry non-cluster noise (leakage, $T_1$
during readout, drift) that limits any mitigation strategy.

\section{Critique (honest)}
\begin{enumerate}
  \item \textbf{Not from paper's specific hardware calibration data.} The noise model
    is a synthetic textbook-cluster instance chosen to match the paper's stated
    structural form (cluster $+$ asymmetric $A_i$ $+$ neighborhood dependence).
    It is \emph{not} reconstructed from IBM Melbourne / Rigetti Aspen's actual
    calibration numbers --- those chips are retired, and while historical
    calibration snapshots may be recoverable from IBM Quantum archives, no such
    ingestion was performed here. Consequence: our absolute numbers cannot be
    compared to Table~1 of the paper on a per-qubit basis --- only the
    qualitative correlated-vs-TP ordering and the order-of-magnitude reduction
    factor are comparable.
  \item \textbf{Comparison against uncorrelated mitigation IS made} (that is
    exactly what the ``tensor-product'' column measures) --- and correlated
    beats TP by $\approx 4\times$ on TVD and $\approx 9\times$ on
    $|\Delta\langle H\rangle|$. That is the substance of Claim C2/C3 and is
    directly exercised.
  \item \textbf{Calibration overhead is NOT quantitatively addressed.} The
    paper's DDOT protocol uses $O(k\cdot 2^k \log N)$ circuits, and the whole
    scalability argument depends on that number. This replication assumes
    perfect characterization ($R_\text{true}$ known exactly) and so bypasses the
    entire characterization cost. A more complete replication would run DDOT
    circuits, estimate $\hat R$, and study how the mitigation quality degrades
    with the number of characterization circuits --- we did not do this.
  \item \textbf{Perfect-characterization limit inflates the reduction factor.}
    On real hardware you mitigate with $\hat R$, not $R_\text{true}$; the
    characterization residual bounds the achievable reduction. Our $30.8\times$
    should therefore be read as an \emph{upper bound} on what the paper's
    method achieves in the small-$N$ cluster regime, not as a direct number
    to compare to hardware.
  \item \textbf{$N=4$ is small.} The paper's advantage grows with system size
    because the mismatch $\|R_\text{true} - \bigotimes_i A_i\|$ grows.
    A convincing scaling study would sweep $N\in\{4,8,15\}$ and record how
    the correlated/TP gap widens --- not attempted here.
  \item \textbf{Finite-shot noise treated implicitly.} We use $10^5$ shots per
    circuit and 25 seeds; explicit per-metric error bars are not reported.
    The bar chart shows means only.
  \item \textbf{No comparison against modern mitigation baselines.} The paper
    predates M3, PEC/PEA, and mid-circuit-measurement-based schemes. No
    head-to-head against these was attempted.
\end{enumerate}

\section{What was NOT reproduced}
\begin{itemize}
  \item Hardware numbers on IBM Melbourne / Rigetti Aspen (chips retired).
  \item DDOT characterization circuit-count scaling with $N$.
  \item MAX-2-SAT / SK-model QAOA benchmarks at $N\gg 4$.
  \item Sensitivity of mitigation quality to imperfect characterization.
\end{itemize}
