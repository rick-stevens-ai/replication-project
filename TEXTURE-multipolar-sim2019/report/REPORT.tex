\begin{abstract}
We rebuild, from scratch, the cubic $j=3/2$ Luttinger-semimetal mean-field superconductivity
of Sim \emph{et al.} and test the headline claim: \emph{for zero quadrupolar order the weak-coupling
mean-field ground state is the time-reversal-breaking d-wave $\Delta_{e_g}=(1,i)=d_{x^2-y^2}+i\,d_{3z^2-r^2}$.}
We construct the four-band Luttinger Hamiltonian (Eq.~1) with the paper's PrBi parameters, form the
even-parity quintet pairing vertices $M_a=\gamma_{45}\gamma_a$ (Eqs.~3--5), and evaluate (A) the
linearized-gap pairing susceptibility per channel and (B) the Bogoliubov--de Gennes (BdG)
condensation-energy quartic that selects among the degenerate $e_g$ doublet.
\begin{tcolorbox}[colback=blue!5,colframe=blue!50]
\textbf{Headline verdict: PARTIAL.} The $e_g$-over-$t_{2g}$ irrep selection is
\textbf{reproduced} ($\lambda_{e_g}=0.897 > \lambda_{t_{2g}}=0.798$, so $e_g$ leads for
$|c_{e_g}|>|c_{t_{2g}}|$), and the strong-coupling drift toward a \emph{real} TR-symmetric $e_g$
d-wave is reproduced. The \emph{specific} weak-coupling selection of the TR-breaking $(1,i)$ state is
\textbf{not resolved} by our single-gap condensation-energy proxy (we obtain the quartic invariant
$q_2<0$, favoring a real nodal $e_g$ state, opposite to the paper's $q_2>0$).
\textbf{Coverage 6/10, Agreement 6/10.}
\end{tcolorbox}
\end{abstract}

\section{The paper's model}
The itinerant Bi $6p$ electrons form a $j=3/2$ Luttinger semimetal,
\begin{equation}
H_0(\mathbf{k}) = c_0 k^2 + \sum_{i=1}^{5} c_i\, d_i(\mathbf{k})\, \gamma_i - \mu ,
\end{equation}
with $d_1=\tfrac{\sqrt3}{2}(k_x^2-k_y^2)$, $d_2=\tfrac12(3k_z^2-k^2)$, $d_3=\sqrt3 k_yk_z$,
$d_4=\sqrt3 k_zk_x$, $d_5=\sqrt3 k_xk_y$, and anti-commuting Dirac matrices
$\gamma_1=\sigma_x\!\otimes\!I,\ \gamma_2=\sigma_z\!\otimes\!\sigma_z,\ \gamma_3=\sigma_z\!\otimes\!\sigma_y,\
\gamma_4=\sigma_z\!\otimes\!\sigma_x,\ \gamma_5=\sigma_y\!\otimes\!I$. Cubic symmetry:
$c_1=c_2\equiv c_{e_g}$, $c_3=c_4=c_5\equiv c_{t_{2g}}$. PrBi parameters (units $a/\pi$):
$c_0=-6,\ c_{e_g}=-2,\ c_{t_{2g}}=-1$ eV, $\mu=-0.6$ eV.

The Fierz-decomposed attractive d-wave channel ($g_{d_a}=-g$, $H_s$ neglected) yields the
even-parity quintet ($j=2$) order parameters
\begin{equation}
\Delta_a=\langle\psi_{-\mathbf{k}}^T\gamma_{45}\gamma_a\psi_{\mathbf{k}}\rangle,\qquad \gamma_{45}=i\gamma_4\gamma_5,
\end{equation}
with $\Delta_{e_g}=(\Delta_1,\Delta_2)=(d_{x^2-y^2},d_{3z^2-r^2})$ and
$\Delta_{t_{2g}}=(\Delta_3,\Delta_4,\Delta_5)=(d_{yz},d_{zx},d_{xy})$. State selection follows the
Ginzburg--Landau free energy $F_{e_g}=r_{e_g}|\Delta_{e_g}|^2+q_1|\Delta_{e_g}|^4
+q_2(\Delta_1\Delta_2^*-\Delta_2\Delta_1^*)^2$: the TR-breaking $(1,i)$ is the ground state iff $q_2>0$.

\section{Method built from scratch}
\begin{enumerate}
\item Constructed the five $\gamma_i$ by Kronecker products; verified the Clifford algebra
$\{\gamma_i,\gamma_j\}=2\delta_{ij}\mathbb{1}$ numerically (passes).
\item Cross-checked $O_{20}=3J_z^2-J^2$ in the $j=3/2$ basis against the reused
\texttt{ollie\_multipolar\_stevens\_landau\_kernel.py} Stevens-operator conventions (matches
$\mathrm{diag}(6,-6,-6,6)$).
\item \textbf{Part A --- pairing susceptibility.} On a $22^3$ $\mathbf{k}$-grid we diagonalize
$H_0(\mathbf{k})$ and evaluate the linearized-gap (Cooper) kernel
$\lambda_a=\langle W(E_s,E_{s'})|\langle u_s(\mathbf{k})|M_a|u_{s'}(\mathbf{k})^*\rangle|^2\rangle_{\mathbf{k}}$
with $M_a=\gamma_{45}\gamma_a$ and $W=(1-f_s-f_{s'})/(E_s+E_{s'})$. The largest $\lambda_a$ is the
leading weak-coupling instability.
\item \textbf{Part B --- $e_g$ selection.} We build the $8\times8$ BdG Hamiltonian
$H_{\mathrm{BdG}}=\left(\begin{smallmatrix}H_0(\mathbf{k}) & \hat\Delta\\ \hat\Delta^\dagger & -H_0(-\mathbf{k})^T\end{smallmatrix}\right)$
with $\hat\Delta=\Delta(\eta_1 M_1+\eta_2 M_2)$, compute the condensation energy relative to the normal
state, and extract the quartic coefficient $B$ (hence $q_2=B_{\rm real}-B_{(1,i)}$) on a Fermi-surface
shell (weak coupling) and over the whole BZ (strong coupling).
\end{enumerate}
Runner: \texttt{\char`~/comfyui-env/bin/python} (numpy 2.3.5). Total runtime $\approx$ 11 s.

\section{Gaps, conventions and scope}
\begin{itemize}
\item \textbf{Primary gap.} The delicate weak-coupling selection of the TR-breaking $(1,i)$ over the
real $(0,1)$ is a \emph{sign-of-}$q_2$ question. Our single-gap BdG condensation-energy proxy gives
$q_2<0$; the TR-breaking state opens Bogoliubov Fermi surfaces (16 gapless pockets) that cost
condensation energy in a whole-Fermi-surface treatment, so the sign flips. Resolving it requires the
exact one-loop GL quartic coefficients with two-band \emph{projected} gaps (Boettcher--Herbut PRL
\textbf{120}, 057002; Sim \emph{et al.} SI Sec.~I), not built here.
\item \textbf{Not built:} the 16-pocket Bogoliubov Fermi surface and its $\pm2$ Chern numbers
(Class~D + inversion, $2\mathbb{Z}$); the full $\tilde J_K\langle O_{20}\rangle$ phase diagram
(Fig.~1) including the $t_{2g}$ $d_{yz}+id_{zx}$ region; the microscopic Kondo-coupling derivation Eq.~(7).
\item \textbf{Conventions:} $g_a=g_1$, attractive $g_{d_a}=-g$, $H_s$ neglected (all per the paper).
\item \textbf{Extraction tooling degraded:} \texttt{marker}/\texttt{nougat} unavailable; extraction via
\texttt{pdftotext}. Equations hand-transcribed here and in \texttt{extraction/nougat.mmd}. Not a physics gap.
\end{itemize}
